% 章標題：CHAPTER N 一行、章名一行，兩行皆全大寫置中，且緊貼版面頂端
\renewcommand\prechaptername{CHAPTER}
\def\@makechapterhead#1{%
  {\parindent \z@ \centering \normalfont
    \ifnum \c@secnumdepth >\m@ne
        {\ntustChapFont \prechaptername \space \thechapter}\par\nobreak
    \fi
    \interlinepenalty\@M
    {\ntustChapFont \MakeUppercase{#1}}\par\nobreak
    \vskip 24\p@
  }}
% 目錄中的章條目同樣全大寫（Harry 送審版的目錄即為大寫）
\def\@chapter[#1]#2{\ifnum \c@secnumdepth >\m@ne
    \refstepcounter{chapter}%
    \typeout{\@chapapp\space\thechapter.}%
    \addcontentsline{toc}{chapter}{\protect\numberline{\thechapter}\MakeUppercase{#1}}%
  \else
    \addcontentsline{toc}{chapter}{\MakeUppercase{#1}}%
  \fi
  \chaptermark{#1}%
  \addtocontents{lof}{\protect\addvspace{10\p@}}%
  \addtocontents{lot}{\protect\addvspace{10\p@}}%
  \if@twocolumn
    \@topnewpage[\@makechapterhead{#2}]%
  \else
    \@makechapterhead{#2}%
    \@afterheading
  \fi}
% 前置頁與參考文獻的無編號章標題：20pt 粗體置中，同樣貼齊版面頂端
